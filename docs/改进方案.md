## 问题1：文件组织混乱

### 🔴 当前问题
- 生成的数据文件（如 `final_3dgs_rl_data\test.pt`）直接存放在项目根目录
- 与源代码、配置文件混在一起，难以管理
- 不同实验的文件可能相互覆盖

### ✅ 解决方案

#### 方案A：时间戳目录结构（推荐）
```python
# 在 ai_lab_repo.py 的 LaboratoryWorkflow.__init__ 中添加
import datetime

def create_experiment_directory(self):
    """创建基于时间戳的实验目录"""
    timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
    topic_short = re.sub(r'[^\w\s-]', '', self.research_topic)[:30]
    topic_short = re.sub(r'[-\s]+', '_', topic_short)
    
    self.experiment_root = f"experiments/{timestamp}_{topic_short}"
    os.makedirs(f"{self.experiment_root}/data", exist_ok=True)
    os.makedirs(f"{self.experiment_root}/outputs", exist_ok=True)
    os.makedirs(f"{self.experiment_root}/logs", exist_ok=True)
    os.makedirs(f"{self.experiment_root}/src", exist_ok=True)
    os.makedirs(f"{self.experiment_root}/reports", exist_ok=True)
    
    return self.experiment_root
```

## 问题2：缺少运行日志记录

### 🔴 当前问题
- 只有 `experiment_output.log` 记录实验执行输出
- 没有记录完整的终端输出（agent对话、调试信息等）
- LLM交互历史没有保存
- 运行完成后无法回溯执行过程

### ✅ 解决方案

#### 方案A：多级日志系统（推荐）

**1. 创建日志管理模块**
```python
# 新建文件：logger_manager.py
import logging
import sys
from datetime import datetime

class AgentLabLogger:
    """统一的日志管理系统"""
    
    def __init__(self, log_dir="logs"):
        self.log_dir = log_dir
        os.makedirs(log_dir, exist_ok=True)
        
        # 1. 完整执行日志（所有输出）
        self.full_logger = self._create_logger(
            'full_execution',
            f'{log_dir}/full_execution.log',
            logging.DEBUG
        )
        
        # 2. LLM交互日志（对话记录）
        self.llm_logger = self._create_logger(
            'llm_interactions',
            f'{log_dir}/llm_interactions.log',
            logging.INFO
        )
        
        # 3. 错误日志（仅错误和警告）
        self.error_logger = self._create_logger(
            'errors',
            f'{log_dir}/errors.log',
            logging.WARNING
        )
        
        # 4. 阶段日志（各阶段完成情况）
        self.phase_logger = self._create_logger(
            'phases',
            f'{log_dir}/phases.log',
            logging.INFO
        )
        
        # 同时输出到终端
        self._add_console_handler()
    
    def _create_logger(self, name, filename, level):
        """创建单个logger"""
        logger = logging.getLogger(name)
        logger.setLevel(level)
        
        # 文件处理器（UTF-8编码支持中文）
        fh = logging.FileHandler(filename, encoding='utf-8')
        fh.setLevel(level)
        
        # 格式化器
        formatter = logging.Formatter(
            '%(asctime)s | %(levelname)s | %(message)s',
            datefmt='%Y-%m-%d %H:%M:%S'
        )
        fh.setFormatter(formatter)
        logger.addHandler(fh)
        
        return logger
    
    def _add_console_handler(self):
        """添加控制台输出"""
        console = logging.StreamHandler(sys.stdout)
        console.setLevel(logging.INFO)
        formatter = logging.Formatter('%(message)s')
        console.setFormatter(formatter)
        self.full_logger.addHandler(console)
    
    def log_llm_interaction(self, model, phase, prompt, response):
        """记录LLM交互"""
        self.llm_logger.info(f"\n{'='*80}")
        self.llm_logger.info(f"模型: {model} | 阶段: {phase}")
        self.llm_logger.info(f"{'='*80}")
        self.llm_logger.info(f"提示词:\n{prompt[:500]}...")
        self.llm_logger.info(f"\n响应:\n{response[:500]}...")
        self.llm_logger.info(f"{'='*80}\n")
    
    def log_phase_start(self, phase_name):
        """记录阶段开始"""
        msg = f"{'*'*50}\n开始阶段: {phase_name}\n{'*'*50}"
        self.full_logger.info(msg)
        self.phase_logger.info(f"开始: {phase_name}")
    
    def log_phase_end(self, phase_name, duration):
        """记录阶段结束"""
        msg = f"阶段 '{phase_name}' 完成，耗时: {duration:.2f}秒"
        self.full_logger.info(msg)
        self.phase_logger.info(f"完成: {phase_name} (耗时: {duration:.2f}s)")
    
    def log_file_saved(self, filepath, filetype):
        """记录文件保存"""
        self.full_logger.info(f"✓ 已保存 {filetype}: {filepath}")
    
    def log_error(self, error_msg, exception=None):
        """记录错误"""
        self.error_logger.error(error_msg)
        if exception:
            self.error_logger.exception(exception)
        self.full_logger.error(error_msg)
```

# 在程序启动时添加
import sys
from datetime import datetime

class TeeOutput:
    """同时输出到终端和文件"""
    def __init__(self, *files):
        self.files = files
    
    def write(self, data):
        for f in self.files:
            f.write(data)
            f.flush()
    
    def flush(self):
        for f in self.files:
            f.flush()

# 启动时创建日志文件
log_file = open(f'logs/full_output_{datetime.now().strftime("%Y%m%d_%H%M%S")}.log', 'w', encoding='utf-8')
sys.stdout = TeeOutput(sys.stdout, log_file)
sys.stderr = TeeOutput(sys.stderr, log_file)

## 问题3：终端输出语言不友好

### 🔴 当前问题
- 所有分隔符和提示信息都是英文
- 阶段标题如 "Beginning phase: literature review" 对中文用户不直观
- 混合中英文阅读体验差

### ✅ 解决方案

#### 方案A：国际化配置文件（推荐）
```python
# 新建文件：i18n_messages.py
MESSAGES = {
    'zh_CN': {
        # 阶段名称
        'literature review': '文献综述',
        'plan formulation': '计划制定',
        'data preparation': '数据准备',
        'running experiments': '运行实验',
        'results interpretation': '结果解读',
        'report writing': '报告撰写',
        'report refinement': '报告完善',
        
        # 提示信息
        'beginning_phase': '开始阶段',
        'beginning_subtask': '开始子任务',
        'phase_completed': '阶段完成',
        'subtask_completed': '子任务完成，耗时',
        'seconds': '秒',
        'saving_file': '正在保存文件',
        'file_saved': '文件已保存',
        'error_occurred': '发生错误',
        
        # 分隔符
        'phase_separator': '*' * 50,
        'subtask_separator': '&' * 30,
        'section_separator': '=' * 80,
    },
    'en_US': {
        'literature review': 'Literature Review',
        'plan formulation': 'Plan Formulation',
        # ... 英文翻译
    }
}

class I18nManager:
    def __init__(self, language='zh_CN'):
        self.language = language
        self.messages = MESSAGES.get(language, MESSAGES['en_US'])
    
    def get(self, key, default=None):
        """获取翻译文本"""
        return self.messages.get(key, default or key)
    
    def format_phase_start(self, phase_name):
        """格式化阶段开始信息"""
        separator = self.messages['phase_separator']
        translated = self.messages.get(phase_name, phase_name)
        return f"\n{separator}\n{self.messages['beginning_phase']}: {translated}\n{separator}"
    
    def format_phase_end(self, phase_name, duration):
        """格式化阶段结束信息"""
        translated = self.messages.get(phase_name, phase_name)
        return f"{self.messages['phase_completed']}: {translated} ({duration:.2f}{self.messages['seconds']})"
```

**在主程序中使用**

# 原：print(f"Subtask '{subtask}' completed in {phase_duration:.2f} seconds.")

**2. 创建配置管理器**

# 新建文件：config_manager.py
import os
import yaml
from dotenv import load_dotenv

class ConfigManager:
    """配置管理器"""
    
    def __init__(self):
        # 加载环境变量
        load_dotenv()
        
        self.config_dir = "experiment_configs"
        self.profiles_dir = f"{self.config_dir}/profiles"
        os.makedirs(self.profiles_dir, exist_ok=True)
        
        # 默认配置
        self.default_config = {
            "copilot-mode": False,
            "language": "Chinese",
            "num-papers-lit-review": 5,
            "num-papers-to-write": 1,
            "mlesolver-max-steps": 3,
            "papersolver-max-steps": 1,
            "compile-latex": False,
            "parallel-labs": False,
            "except-if-fail": False,
        }
    
    def list_profiles(self):
        """列出所有配置文件"""
        profiles = []
        for f in os.listdir(self.profiles_dir):
            if f.endswith('.yaml'):
                profiles.append(f[:-5])  # 去掉.yaml后缀
        return profiles
    
    def load_profile(self, profile_name):
        """加载配置文件"""
        profile_path = f"{self.profiles_dir}/{profile_name}.yaml"
        if os.path.exists(profile_path):
            with open(profile_path, 'r', encoding='utf-8') as f:
                config = yaml.safe_load(f)
        else:
            config = self.default_config.copy()
        
        # 从环境变量加载API密钥
        config['api-key'] = os.getenv('OPENAI_API_KEY')
        config['deepseek-api-key'] = os.getenv('DEEPSEEK_API_KEY')
        config['gitee-api-key'] = os.getenv('GITEE_API_KEY')
        config['gitee-base-url'] = os.getenv('GITEE_BASE_URL', 'https://ai.gitee.com/v1')
        
        return config
    
    def save_profile(self, profile_name, config):
        """保存配置文件"""
        # 不保存API密钥到文件
        safe_config = config.copy()
        safe_config.pop('api-key', None)
        safe_config.pop('deepseek-api-key', None)
        safe_config.pop('gitee-api-key', None)
        
        profile_path = f"{self.profiles_dir}/{profile_name}.yaml"
        with open(profile_path, 'w', encoding='utf-8') as f:
            yaml.dump(safe_config, f, allow_unicode=True, default_flow_style=False)
    
    def create_quick_config(self, research_topic, model="DeepSeek-V3", api_provider="gitee"):
        """快速创建配置"""
        config = self.default_config.copy()
        config['research-topic'] = research_topic
        config['llm-backend'] = model
        
        # 根据provider设置API密钥
        if api_provider == "gitee":
            config['gitee-api-key'] = os.getenv('GITEE_API_KEY')
            config['gitee-base-url'] = os.getenv('GITEE_BASE_URL', 'https://ai.gitee.com/v1')
        elif api_provider == "deepseek":
            config['deepseek-api-key'] = os.getenv('DEEPSEEK_API_KEY')
        elif api_provider == "openai":
            config['api-key'] = os.getenv('OPENAI_API_KEY')
        
        return config

class TopicManager:
    """研究主题管理器"""
    
    def __init__(self, history_file="topic_history.json"):
        self.history_file = history_file
        self.templates = TOPIC_TEMPLATES
        self.history = self._load_history()
    
    def _load_history(self):
        """加载历史主题"""
        if os.path.exists(self.history_file):
            with open(self.history_file, 'r', encoding='utf-8') as f:
                return json.load(f)
        return []
    
    def _save_history(self):
        """保存历史主题"""
        with open(self.history_file, 'w', encoding='utf-8') as f:
            json.dump(self.history, f, ensure_ascii=False, indent=2)
    
    def list_templates(self):
        """列出所有模板"""
        return [
            {
                "id": key,
                "name": template["name"],
                "template": template["template"]
            }
            for key, template in self.templates.items()
        ]
    
    def create_from_template(self, template_id, **kwargs):
        """从模板创建主题"""
        if template_id not in self.templates:
            raise ValueError(f"模板 {template_id} 不存在")
        
        template = self.templates[template_id]
        topic = template["template"].format(**kwargs)
        
        # 获取相关的task notes
        task_notes = template.get("task_notes", {})
        
        # 保存到历史
        self.history.insert(0, {
            "topic": topic,
            "template_id": template_id,
            "created_at": datetime.now().isoformat(),
            "variables": kwargs
        })
        self._save_history()
        
        return topic, task_notes
    
    def get_history(self, limit=10):
        """获取历史主题"""
        return self.history[:limit]
    
    def search_history(self, keyword):
        """搜索历史主题"""
        return [
            item for item in self.history
            if keyword.lower() in item["topic"].lower()
        ]
